cmake_minimum_required(VERSION 3.22.1)
project(NUNAndroidBridge LANGUAGES C)
set(NUN_BUILD_TESTS OFF CACHE BOOL "Disable host tests for Android" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Disable imported test executables" FORCE)
get_filename_component(NUN_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../core/NUN" ABSOLUTE)
if(NOT EXISTS "${NUN_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "NUN source submodule is missing. Run git submodule update --init --recursive.")
endif()
add_subdirectory("${NUN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/nun-core")
add_library(nun_android_bridge SHARED native_bridge.c)
target_link_libraries(nun_android_bridge PRIVATE NUN.Runtime log)
target_compile_features(nun_android_bridge PRIVATE c_std_11)
target_compile_options(nun_android_bridge PRIVATE -Wall -Wextra -Wpedantic)
